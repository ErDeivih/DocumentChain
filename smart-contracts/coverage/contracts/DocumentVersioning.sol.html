// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;
 
import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";
 
/**
 * @title DocumentVersioning
 * @dev Gestión de versiones de documentos
 */
contract DocumentVersioning is ReentrancyGuard {
    struct Version {
        bytes32 docId;
        uint256 versionNumber;
        string ipfsCid;
        address createdBy;
        uint256 createdAt;
        bool isOperational;       // Solo una versión puede ser operativa
        uint256 restoredFrom;     // Si fue restaurada (0 = no restaurada)
    }
    
    mapping(bytes32 => Version[]) public documentVersions;
    mapping(bytes32 => uint256) public latestVersion;
    mapping(bytes32 => uint256) public operationalVersion;
    
    event VersionCreated(
        bytes32 indexed docId,
        uint256 versionNumber,
        string ipfsCid,
        address indexed createdBy,
        uint256 timestamp
    );
    
    event VersionRestored(
        bytes32 indexed docId,
        uint256 newVersionNumber,
        uint256 restoredFromVersion,
        string ipfsCid,
        address indexed by,
        uint256 timestamp
    );
    
    event VersionSetOperational(
        bytes32 indexed docId,
        uint256 versionNumber,
        address indexed by,
        uint256 timestamp
    );
    
    /**
     * @dev Inicializar documento con primera versión
     */
    function initializeDocument(
        bytes32 _docId,
        string calldata _ipfsCid
    ) external {
        require(latestVersion[_docId] == 0, "Document already initialized");
        require(bytes(_ipfsCid).length > 0, "Invalid IPFS CID");
        
        documentVersions[_docId].push(Version({
            docId: _docId,
            versionNumber: 1,
            ipfsCid: _ipfsCid,
            createdBy: msg.sender,
            createdAt: block.timestamp,
            isOperational: true,
            restoredFrom: 0
        }));
        
        latestVersion[_docId] = 1;
        operationalVersion[_docId] = 1;
        
        emit VersionCreated(_docId, 1, _ipfsCid, msg.sender, block.timestamp);
        emit VersionSetOperational(_docId, 1, msg.sender, block.timestamp);
    }
    
    /**
     * @dev Crear nueva versión
     */
    function createVersion(
        bytes32 _docId,
        string calldata _ipfsCid
    ) external EnonReentrant {
        require(operationalVersion[_docId] > 0, "Document not initialized");
        Erequire(latestVersion[_docId] > 0, "Document not found");
        require(bytes(_ipfsCid).length > 0, "Invalid IPFS CID");
        
        uint256 newVersionNumber = latestVersion[_docId] + 1;
        
        documentVersions[_docId].push(Version({
            docId: _docId,
            versionNumber: newVersionNumber,
            ipfsCid: _ipfsCid,
            createdBy: msg.sender,
            createdAt: block.timestamp,
            isOperational: true,
            restoredFrom: 0
        }));
        
        // Marcar la anterior como no operativa
        _setNonOperational(_docId, operationalVersion[_docId]);
        
        latestVersion[_docId] = newVersionNumber;
        operationalVersion[_docId] = newVersionNumber;
        
        emit VersionCreated(_docId, newVersionNumber, _ipfsCid, msg.sender, block.timestamp);
        emit VersionSetOperational(_docId, newVersionNumber, msg.sender, block.timestamp);
    }
    
    /**
     * @dev Restaurar versión anterior (crea nueva versión con contenido de la anterior)
     */
    function restoreVersion(
        bytes32 _docId,
        uint256 _versionToRestore
    ) external EnonReentrant {
        require(operationalVersion[_docId] > 0, "Document not initialized");
        require(_versionToRestore <= latestVersion[_docId], "Invalid version");
        require(_versionToRestore > 0, "Invalid version");
        
        // Buscar la versión a restaurar
        string memory ipfsCid;
        bool found = false;
        
        for (uint i = 0; i < documentVersions[_docId].length; i++) {
            if (documentVersions[_docId][i].versionNumber == _versionToRestore) {
                ipfsCid = documentVersions[_docId][i].ipfsCid;
                found = true;
                break;
            }
        }
        
        Erequire(found, "Version not found");
        Erequire(bytes(ipfsCid).length > 0, "Invalid IPFS CID");
        
        uint256 newVersionNumber = latestVersion[_docId] + 1;
        
        documentVersions[_docId].push(Version({
            docId: _docId,
            versionNumber: newVersionNumber,
            ipfsCid: ipfsCid,  // Usar el CID de la versión restaurada
            createdBy: msg.sender,
            createdAt: block.timestamp,
            isOperational: true,
            restoredFrom: _versionToRestore
        }));
        
        // Marcar la anterior como no operativa
        _setNonOperational(_docId, operationalVersion[_docId]);
        
        latestVersion[_docId] = newVersionNumber;
        operationalVersion[_docId] = newVersionNumber;
        
        emit VersionRestored(_docId, newVersionNumber, _versionToRestore, ipfsCid, msg.sender, block.timestamp);
        emit VersionSetOperational(_docId, newVersionNumber, msg.sender, block.timestamp);
    }
    
    /**
     * @dev Obtener todas las versiones de un documento
     */
    function getVersions(bytes32 _docId) external view returns (Version[] memory) {
        return documentVersions[_docId];
    }
    
    /**
     * @dev Obtener versión operativa
     */
    function getOperationalVersion(bytes32 _docId) external view returns (Version memory) {
        uint256 opVer = operationalVersion[_docId];
        require(opVer > 0, "No operational version");
        
        for (uint i = 0; i < documentVersions[_docId].length; i++) {
            if (documentVersions[_docId][i].versionNumber == opVer) {
                return documentVersions[_docId][i];
            }
        }
        
        revert("Operational version not found");
    }
    
    /**
     * @dev Obtener versión específica
     */
    function getVersion(bytes32 _docId, uint256 _versionNumber) external view returns (Version memory) {
        for (uint i = 0; i < documentVersions[_docId].length; i++) {
            if (documentVersions[_docId][i].versionNumber == _versionNumber) {
                return documentVersions[_docId][i];
            }
        }
        
        revert("Version not found");
    }
    
    /**
     * @dev Obtener número de versión operativa
     */
    function getOperationalVersionNumber(bytes32 _docId) external view returns (uint256) {
        return operationalVersion[_docId];
    }
    
    /**
     * @dev Obtener número de última versión
     */
    function getLatestVersionNumber(bytes32 _docId) external view returns (uint256) {
        return latestVersion[_docId];
    }
    
    /**
     * @dev Marcar versión como no operativa (internal)
     */
    function _setNonOperational(bytes32 _docId, uint256 _versionNumber) internal {
        for (uint i = 0; i < documentVersions[_docId].length; i++) {
            if (documentVersions[_docId][i].versionNumber == _versionNumber) {
                documentVersions[_docId][i].isOperational = false;
                break;
            }
        }
    }
}
