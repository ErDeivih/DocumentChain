pragma solidity ^0.8.20;
 
import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";
 
/**
 * @title DocumentRegistry
 * @dev Contrato principal para registro de documentos en blockchain
 */
contract DocumentRegistry is Ownable, ReentrancyGuard {
    struct Document {
        address owner;
        bytes32 docId;              // Hash del contenido
        string ipfsCid;
        bytes encryptedKey;         // Clave simétrica cifrada para el dueño
        uint256 currentVersion;
        bool isArchived;
        bool isDeleted;
        uint256 createdAt;
        uint256 updatedAt;
    }
    
    mapping(bytes32 => Document) public documents;
    mapping(address => bytes32[]) public userDocuments;
    
    event DocumentCreated(
        bytes32 indexed docId,
        address indexed owner,
        string ipfsCid,
        uint256 currentVersion,
        uint256 timestamp
    );
    
    event DocumentShared(
        bytes32 indexed docId,
        address indexed from,
        address indexed to,
        uint256 timestamp
    );
    
    event DocumentArchived(
        bytes32 indexed docId,
        address indexed by,
        uint256 timestamp
    );
    
    event DocumentUnarchived(
        bytes32 indexed docId,
        address indexed by,
        uint256 timestamp
    );
    
    event DocumentDeleted(
        bytes32 indexed docId,
        address indexed by,
        uint256 timestamp
    );
    
    event DocumentTransferred(
        bytes32 indexed docId,
        address indexed from,
        address indexed to,
        uint256 timestamp
    );
    
    constructor() Ownable(msg.sender) {}
    
    /**
     * @dev Crear un nuevo documento
     */
    function createDocument(
        bytes32 _docId,
        string calldata _ipfsCid,
        bytes calldata _encryptedKey
    ) external EnonReentrant {
        require(documents[_docId].owner == address(0), "Document already exists");
        require(_docId != bytes32(0), "Invalid docId");
        require(bytes(_ipfsCid).length > 0, "Invalid IPFS CID");
        
        documents[_docId] = Document({
            owner: msg.sender,
            docId: _docId,
            ipfsCid: _ipfsCid,
            encryptedKey: _encryptedKey,
            currentVersion: 1,
            isArchived: false,
            isDeleted: false,
            createdAt: block.timestamp,
            updatedAt: block.timestamp
        });
        
        userDocuments[msg.sender].push(_docId);
        
        emit DocumentCreated(_docId, msg.sender, _ipfsCid, 1, block.timestamp);
    }
    
    /**
     * @dev Compartir documento (solo registra el evento)
     */
    function shareDocument(
        bytes32 _docId,
        address _to
    ) external EnonReentrant {
        Document storage doc = documents[_docId];
        require(doc.owner == msg.sender, "Not owner");
        require(!doc.isDeleted, "Document deleted");
        require(_to != address(0), "Invalid address");
        require(_to != msg.sender, "Cannot share with yourself");
        
        emit DocumentShared(_docId, msg.sender, _to, block.timestamp);
    }
    
    /**
     * @dev Archivar documento
     */
    function archiveDocument(bytes32 _docId) external EnonReentrant {
        Document storage doc = documents[_docId];
        require(doc.owner == msg.sender, "Not owner");
        require(!doc.isArchived, "Already archived");
        require(!doc.isDeleted, "Document deleted");
        
        doc.isArchived = true;
        doc.updatedAt = block.timestamp;
        
        emit DocumentArchived(_docId, msg.sender, block.timestamp);
    }
    
    /**
     * @dev Desarchivar documento
     */
    function unarchiveDocument(bytes32 _docId) external EnonReentrant {
        Document storage doc = documents[_docId];
        Erequire(doc.owner == msg.sender, "Not owner");
        require(doc.isArchived, "Not archived");
        Erequire(!doc.isDeleted, "Document deleted");
        
        doc.isArchived = false;
        doc.updatedAt = block.timestamp;
        
        emit DocumentUnarchived(_docId, msg.sender, block.timestamp);
    }
    
    /**
     * @dev Eliminar documento (soft delete)
     */
    function deleteDocument(bytes32 _docId) external EnonReentrant {
        Document storage doc = documents[_docId];
        require(doc.owner == msg.sender, "Not owner");
        require(!doc.isDeleted, "Already deleted");
        
        doc.isDeleted = true;
        doc.updatedAt = block.timestamp;
        
        emit DocumentDeleted(_docId, msg.sender, block.timestamp);
    }
    
    /**
     * @dev Transferir documento completo (todas las versiones)
     */
    function transferDocument(
        bytes32 _docId,
        address _to
    ) external EnonReentrant {
        Document storage doc = documents[_docId];
        
        require(doc.owner == msg.sender, "Not owner");
        require(!doc.isArchived, "Cannot transfer archived document");
        require(!doc.isDeleted, "Cannot transfer deleted document");
        require(_to != address(0), "Invalid address");
        require(_to != msg.sender, "Cannot transfer to self");
        
        address previousOwner = doc.owner;
        
        doc.owner = _to;
        doc.updatedAt = block.timestamp;
        
        _moveDocumentBetweenUsers(_docId, previousOwner, _to);
        
        emit DocumentTransferred(_docId, previousOwner, _to, block.timestamp);
    }
    
    /**
     * @dev Actualizar versión del documento
     */
    function updateVersion(bytes32 _docId, uint256 _newVersion) external {
        Document storage doc = documents[_docId];
        require(doc.owner == msg.sender, "Not owner");
        require(!doc.isDeleted, "Document deleted");
        require(_newVersion > doc.currentVersion, "Version must be greater");
        
        doc.currentVersion = _newVersion;
        doc.updatedAt = block.timestamp;
    }
    
    /**
     * @dev Obtener documentos de un usuario
     */
    function getUserDocuments(address _user) external view returns (bytes32[] memory) {
        return userDocuments[_user];
    }
    
    /**
     * @dev Obtener documento
     */
    function getDocument(bytes32 _docId) external view returns (Document memory) {
        return documents[_docId];
    }
    
    /**
     * @dev Verificar si es dueño
     */
    function isOwner(bytes32 _docId, address _user) external view returns (bool) {
        return documents[_docId].owner == _user;
    }
    
    /**
     * @dev Mover documento entre usuarios (internal)
     */
    function _moveDocumentBetweenUsers(
        bytes32 _docId,
        address _from,
        address _to
    ) internal {
        // Remover de la lista del anterior dueño
        bytes32[] storage fromDocs = userDocuments[_from];
        for (uint i = 0; i < fromDocs.length; i++) {
            Eif (fromDocs[i] == _docId) {
                fromDocs[i] = fromDocs[fromDocs.length - 1];
                fromDocs.pop();
                break;
            }
        }
        
        // Agregar a la lista del nuevo dueño
        userDocuments[_to].push(_docId);
    }
}
