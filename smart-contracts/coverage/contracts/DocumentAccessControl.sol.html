// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;
 
import "@openzeppelin/contracts/access/AccessControl.sol";
import "@openzeppelin/contracts/access/Ownable.sol";
 
/**
 * @title DocumentAccessControl
 * @dev Control de acceso a documentos usando roles de OpenZeppelin
 */
contract DocumentAccessControl is AccessControl, Ownable {
    bytes32 public constant ADMIN_ROLE = keccak256("ADMIN_ROLE");
    
    // Roles de documento
    bytes32 public constant DOCUMENT_OWNER = keccak256("DOCUMENT_OWNER");
    bytes32 public constant DOCUMENT_SHARED_READ = keccak256("DOCUMENT_SHARED_READ");
    bytes32 public constant DOCUMENT_SHARED_WRITE = keccak256("DOCUMENT_SHARED_WRITE");
    bytes32 public constant DOCUMENT_SHARED_ADMIN = keccak256("DOCUMENT_SHARED_ADMIN");
    
    mapping(bytes32 => address) public documentOwners;
    mapping(bytes32 => mapping(address => bytes32)) public documentRoles;
    
    event PermissionGranted(
        bytes32 indexed docId,
        address indexed user,
        bytes32 indexed role,
        uint256 timestamp
    );
    
    event PermissionRevoked(
        bytes32 indexed docId,
        address indexed user,
        uint256 timestamp
    );
    
    event DocumentOwnerChanged(
        bytes32 indexed docId,
        address indexed from,
        address indexed to,
        uint256 timestamp
    );
    
    constructor() Ownable(msg.sender) {
        _grantRole(DEFAULT_ADMIN_ROLE, msg.sender);
        _grantRole(ADMIN_ROLE, msg.sender);
    }
    
    /**
     * @dev Crear documento con dueño
     */
    function createDocument(bytes32 _docId, address _owner) external {
        require(documentOwners[_docId] == address(0), "Document already exists");
        require(_owner != address(0), "Invalid owner");
        require(_docId != bytes32(0), "Invalid docId");
        
        documentOwners[_docId] = _owner;
        documentRoles[_docId][_owner] = DOCUMENT_OWNER;
        
        emit PermissionGranted(_docId, _owner, DOCUMENT_OWNER, block.timestamp);
    }
    
    /**
     * @dev Cambiar dueño (usado en transferencia)
     */
    function changeDocumentOwner(bytes32 _docId, address _newOwner) external {
        require(
            msg.sender == documentOwners[_docId] || hasRole(ADMIN_ROLE, msg.sender), 
            "Not authorized"
        );
        require(_newOwner != address(0), "Invalid address");
        require(_newOwner != documentOwners[_docId], "Same owner");
        
        address previousOwner = documentOwners[_docId];
        
        // Eliminar rol de dueño anterior
        delete documentRoles[_docId][previousOwner];
        
        // Asignar nuevo dueño
        documentOwners[_docId] = _newOwner;
        documentRoles[_docId][_newOwner] = DOCUMENT_OWNER;
        
        emit PermissionRevoked(_docId, previousOwner, block.timestamp);
        emit PermissionGranted(_docId, _newOwner, DOCUMENT_OWNER, block.timestamp);
        emit DocumentOwnerChanged(_docId, previousOwner, _newOwner, block.timestamp);
    }
    
    /**
     * @dev Asignar permiso (solo dueño)
     */
    function grantPermission(
        bytes32 _docId,
        address _user,
        bytes32 _role
    ) external {
        require(msg.sender == documentOwners[_docId], "Not document owner");
        require(_user != address(0), "Invalid address");
        require(
            _role == DOCUMENT_SHARED_READ ||
            _role == DOCUMENT_SHARED_WRITE ||
            _role == DOCUMENT_SHARED_ADMIN,
            "Invalid role"
        );
        require(documentRoles[_docId][_user] == bytes32(0), "Already has role");
        
        documentRoles[_docId][_user] = _role;
        
        emit PermissionGranted(_docId, _user, _role, block.timestamp);
    }
    
    /**
     * @dev Revocar permiso (solo dueño)
     */
    function revokePermission(bytes32 _docId, address _user) external {
        require(msg.sender == documentOwners[_docId], "Not document owner");
        require(_user != documentOwners[_docId], "Cannot revoke owner");
        
        delete documentRoles[_docId][_user];
        
        emit PermissionRevoked(_docId, _user, block.timestamp);
    }
    
    /**
     * @dev Verificar si tiene permiso específico
     */
    function hasPermission(
        bytes32 _docId,
        address _user,
        bytes32 _role
    ) external view returns (bool) {
        return documentRoles[_docId][_user] == _role;
    }
    
    /**
     * @dev Verificar si puede leer
     */
    function canRead(bytes32 _docId, address _user) external view returns (bool) {
        bytes32 role = documentRoles[_docId][_user];
        return role == DOCUMENT_OWNER ||
               role == DOCUMENT_SHARED_READ ||
               role == DOCUMENT_SHARED_WRITE ||
               role == DOCUMENT_SHARED_ADMIN;
    }
    
    /**
     * @dev Verificar si puede escribir
     */
    function canWrite(bytes32 _docId, address _user) external view returns (bool) {
        bytes32 role = documentRoles[_docId][_user];
        return role == DOCUMENT_OWNER ||
               role == DOCUMENT_SHARED_WRITE ||
               role == DOCUMENT_SHARED_ADMIN;
    }
    
    /**
     * @dev Verificar si puede compartir
     */
    function canShare(bytes32 _docId, address _user) external view returns (bool) {
        bytes32 role = documentRoles[_docId][_user];
        return role == DOCUMENT_OWNER ||
               role == DOCUMENT_SHARED_ADMIN;
    }
    
    /**
     * @dev Verificar si es dueño
     */
    function isOwner(bytes32 _docId, address _user) external view returns (bool) {
        return documentOwners[_docId] == _user;
    }
    
    /**
     * @dev Obtener rol de un usuario
     */
    function getUserRole(bytes32 _docId, address _user) external view returns (bytes32) {
        return documentRoles[_docId][_user];
    }
    
    /**
     * @dev Obtener dueño del documento
     */
    function getDocumentOwner(bytes32 _docId) external view returns (address) {
        return documentOwners[_docId];
    }
}
