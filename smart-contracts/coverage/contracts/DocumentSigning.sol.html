// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;
 
/**
 * @title DocumentSigning
 * @dev Gestión de firmas digitales de documentos
 */
contract DocumentSigning {
    struct Signature {
        bytes32 docId;
        uint256 versionNumber;
        address signer;
        bytes signature;
        string message;         // Mensaje firmado
        string comment;         // Comentario breve sobre la firma
        uint256 timestamp;
    }
    
    // Mapping: docId → version → signatures
    mapping(bytes32 => mapping(uint256 => Signature[])) public versionSignatures;
    mapping(bytes32 => mapping(uint256 => mapping(address => bool))) public hasSignedVersion;
    
    event DocumentVersionSigned(
        bytes32 indexed docId,
        uint256 indexed versionNumber,
        address indexed signer,
        string message,
        string comment,
        uint256 timestamp
    );
    
    /**
     * @dev Firmar una versión específica de un documento
     */
    function signDocumentVersion(
        bytes32 _docId,
        uint256 _versionNumber,
        bytes calldata _signature,
        string calldata _message,
        string calldata _comment
    ) external {
        require(_docId != bytes32(0), "Invalid docId");
        require(_versionNumber > 0, "Invalid version number");
        require(_signature.length > 0, "Invalid signature");
        require(bytes(_message).length > 0, "Invalid message");
        
        // Verificar que no haya firmado esta versión antes
        require(
            !hasSignedVersion[_docId][_versionNumber][msg.sender],
            "Already signed this version"
        );
        
        // Registrar la firma
        versionSignatures[_docId][_versionNumber].push(Signature({
            docId: _docId,
            versionNumber: _versionNumber,
            signer: msg.sender,
            signature: _signature,
            message: _message,
            comment: _comment,
            timestamp: block.timestamp
        }));
        
        hasSignedVersion[_docId][_versionNumber][msg.sender] = true;
        
        emit DocumentVersionSigned(_docId, _versionNumber, msg.sender, _message, _comment, block.timestamp);
    }
    
    /**
     * @dev Obtener firmas de una versión específica
     */
    function getVersionSignatures(
        bytes32 _docId,
        uint256 _versionNumber
    ) external view returns (Signature[] memory) {
        return versionSignatures[_docId][_versionNumber];
    }
    
    /**
     * @dev Obtener número de firmas de una versión
     */
    function getVersionSignatureCount(
        bytes32 _docId,
        uint256 _versionNumber
    ) external view returns (uint256) {
        return versionSignatures[_docId][_versionNumber].length;
    }
    
    /**
     * @dev Verificar si un usuario ha firmado una versión
     */
    function hasUserSignedVersion(
        bytes32 _docId,
        uint256 _versionNumber,
        address _user
    ) external view returns (bool) {
        return hasSignedVersion[_docId][_versionNumber][_user];
    }
    
    /**
     * @dev Obtener todas las versiones con firmas (limitado a 100 versiones)
     */
    function getAllDocumentSignatures(
        bytes32 _docId
    ) external view returns (
        uint256[] memory versions,
        uint256[] memory signatureCounts
    ) {
        uint256 count = 0;
        
        // Contar versiones con firmas
        for (uint256 i = 1; i <= 100; i++) {
            if (versionSignatures[_docId][i].length > 0) {
                count++;
            }
        }
        
        versions = new uint256[](count);
        signatureCounts = new uint256[](count);
        
        uint256 index = 0;
        for (uint256 i = 1; i <= 100; i++) {
            if (versionSignatures[_docId][i].length > 0) {
                versions[index] = i;
                signatureCounts[index] = versionSignatures[_docId][i].length;
                index++;
            }
        }
        
        return (versions, signatureCounts);
    }
}
